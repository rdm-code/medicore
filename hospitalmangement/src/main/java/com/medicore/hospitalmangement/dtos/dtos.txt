public record DoctorCreateDTO(
        @NotBlank(message = "Email is required")
        @Email(message = "Email must be valid")
        String email,

        @NotBlank(message = "Password is required")
        @Size(min = 6, max = 20, message = "Password must be 6 to 20 characters")
        String password,

        @NotBlank(message = "Full name is required")
        @Size(max = 100, message = "Full name must be at most 100 characters")
        String fullName,

        @NotBlank(message = "Specialization is required")
        @Size(max = 100, message = "Specialization must be at most 100 characters")
        String specialization,

        @Pattern(regexp = "^[6-9]\\d{9}$",
                message = "Phone must be a valid 10-digit Indian mobile number")
        String phone
) {}

public record PatientCreateDTO(

        // ---- User (login) fields ----
        @NotBlank(message = "Email is required")
        @Email(message = "Email must be valid")
        String email,

        @NotBlank(message = "Password is required")
        @Size(min = 6, max = 20, message = "Password must be 6 to 20 characters")
        String password,

        // ---- Patient profile fields ----
        @NotBlank(message = "Full name is required")
        @Size(max = 100, message = "Full name must be at most 100 characters")
        String fullName,

        @NotNull(message = "Age is required")
        @Min(value = 0, message = "Age cannot be negative")
        @Max(value = 130, message = "Age must be realistic")
        Integer age,

        @NotBlank(message = "Gender is required")
        @Pattern(regexp = "(?i)MALE|FEMALE|OTHER",
                message = "Gender must be MALE, FEMALE or OTHER")
        String gender,

        @Pattern(regexp = "^[6-9]\\d{9}$",
                message = "Phone must be a valid 10-digit Indian mobile number")
        String phone
) {}

public record UserCreateDTO(
        @NotBlank(message = "Email is required")

        String email,

        @NotBlank(message = "Password is required")
        @Size(min = 6, message = "Password must be at least 6 characters long")
        String password,

        String displayName,

        @NotNull(message = "Role is required")
        @Pattern(regexp = "ADMIN|DOCTOR|PATIENT", message = "Role must be either ADMIN or DOCTOR or PATIENT")
        String role
) {}
