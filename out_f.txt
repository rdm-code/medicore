  const { appointments, isLoading, error, complete, addNotes, invalidate } = useMyAppointments();

     },

    onSubmit: async (values) => {
      setServerError('');
      try {
        if (!editing) return;
        await update(editing.id, {
          fullName: values.fullName,
          phone: values.phone ?? '',
          age: typeof values.age === 'number' ? values.age : undefined,
          gender: (values.gender || editing.gender || 'MALE').toUpperCase() as 'MALE' | 'FEMALE' | 'OTHER',

export function toLocalDateTimeString(date: string, time: string): string {
  // date = "YYYY-MM-DD", time = "HH:MM"
  return `${date}T${time.length === 5 ? `${time}:00` : time}`;
}

export function formatDateTime(iso: string): string {
  const d = new Date(iso);
  if (isNaN(d.getTime())) return iso;
  return d.toLocaleString(undefined, {
    dateStyle: "medium",
    timeStyle: "short",
  });
}

export function formatDate(iso: string): string {
  const d = new Date(iso);
  if (isNaN(d.getTime())) return iso;
  return d.toLocaleDateString(undefined, { dateStyle: "medium" });
}

export function formatTime(iso: string): string {
  const d = new Date(iso);
  if (isNaN(d.getTime())) return iso;
  return d.toLocaleTimeString(undefined, {
    hour: "2-digit",
    minute: "2-digit",
  });
}

/** For <input type="date"> from an ISO LocalDateTime */
export function isoDatePart(iso: string): string {
  return iso.slice(0, 10);
}

/** For <input type="time"> from an ISO LocalDateTime */
export function isoTimePart(iso: string): string {
  return iso.slice(11, 16);
}
import { ZodError, ZodType } from "zod";

export type FieldErrors<T> = Partial<Record<keyof T, string>>;

export function zodToFieldErrors<T>(error: ZodError<T>): FieldErrors<T> {
  const result: FieldErrors<T> = {};
  for (const issue of error.issues) {
    const path = issue.path[0] as keyof T | undefined;
    if (path !== undefined && !result[path]) {
      result[path] = issue.message;
    }
  }
  return result;
}

/**
 * Validate an arbitrary input with a Zod schema.
 * Returns either `{ data }` or `{ errors }`.
 */
export function validate<T>(
  schema: ZodType<T>,
  input: unknown,
): { success: true; data: T } | { success: false; errors: FieldErrors<T> } {
  const parsed = schema.safeParse(input);
  if (parsed.success) return { success: true, data: parsed.data };
  return { success: false, errors: zodToFieldErrors(parsed.error) };
}

// ---------- Basic credentials (session storage) ----------
const CRED_KEY = "medicore_basic";

export function setBasicCredentials(email: string, password: string) {
  sessionStorage.setItem(CRED_KEY, btoa(`${email}:${password}`));
}

export function clearBasicCredentials() {
  sessionStorage.removeItem(CRED_KEY);
}

function getBasicHeader(): string | null {
  const raw = sessionStorage.getItem(CRED_KEY);
  return raw ? `Basic ${raw}` : null;
}

// ---------- Errors ----------
export class HttpError extends Error {
  status: number;
  details?: unknown;
  constructor(status: number, message: string, details?: unknown) {
    super(message);
    this.status = status;
    this.details = details;
  }
}

export class ResponseValidationError extends Error {
  zodError: z.ZodError;
  raw: unknown;
  constructor(zodError: z.ZodError, raw: unknown) {
    super("Server returned an unexpected response shape");
    this.zodError = zodError;
    this.raw = raw;
  }
}

// ---------- Axios instance ----------
const http: AxiosInstance = axios.create({
  baseURL: BASE_URL,
  withCredentials: true,
  headers: { "Content-Type": "application/json" },
});

http.interceptors.request.use((config) => {
  if (config.headers && config.headers.Authorization === "") {
    delete config.headers.Authorization;
  } else {
    const basic = getBasicHeader();
    if (basic) config.headers.Authorization = basic;
  }
  return config;
});

http.interceptors.response.use(
  (res) => res,
  (error: AxiosError) => {
    if (error.response) {
      const data = error.response.data as
        | { message?: string; error?: string }
        | undefined;
      const message = data?.message ?? data?.error ?? error.response.statusText;
      throw new HttpError(error.response.status, message, data);
    }
    throw new HttpError(0, error.message);
  },
);

export const authApi = {
  /** POST /api/auth/login */
  async login(payload: LoginRequestDTO): Promise<CurrentUser> {
    setBasicCredentials(payload.email, payload.password);
    try {
      const res = await http.post("/api/auth/login", payload);
      const parsed = authCurrentUserSchema.safeParse(res.data);
      if (!parsed.success) {
        console.error(
          "login: response validation failed",
          parsed.error,
          res.data,
        );
        throw new ResponseValidationError(parsed.error, res.data);
      }
      return parsed.data as CurrentUser;
    } catch (e) {
      clearBasicCredentials();
      throw e;
    }
  },

  /** POST /api/auth/register */
  async register(payload: PatientCreateDTO): Promise<void> {
    await http.post("/api/auth/register", payload, {
      headers: { Authorization: "" },
    });
  },

  /** GET /api/auth/me */
  async me(): Promise<CurrentUser> {
    const res = await http.get("/api/auth/me");
    const parsed = authCurrentUserSchema.safeParse(res.data);
    if (!parsed.success) {
      console.error("me: response validation failed", parsed.error, res.data);
      throw new ResponseValidationError(parsed.error, res.data);
    }
    return parsed.data as CurrentUser;
  },

  /** PUT /api/auth/change-password */
  async changePassword(payload: ChangePasswordPayload): Promise<void> {
    await http.put("/api/auth/change-password", payload);
  },

  /** POST /logout */
  async logout(): Promise<void> {
    clearBasicCredentials();
    try {
      await http.post("/api/auth/logout");
    } catch {
      /* ignore */
    }
  },
};

import { useState, useCallback, useMemo } from "react";
import type { ZodType } from "zod";
import { zodToFieldErrors, type FieldErrors } from "../lib/validation";

export interface UseZodFormOptions<T> {
  initialValues: T;
  schema: ZodType<T>;
  onSubmit: (values: T) => Promise<void> | void;
}

export interface UseZodFormResult<T> {
  values: T;
  errors: FieldErrors<T>;
  touched: Partial<Record<keyof T, boolean>>;
  submitting: boolean;
  serverError: string;
  setValue: <K extends keyof T>(field: K, value: T[K]) => void;
  setTouched: (field: keyof T) => void;
  handleChange: (
    e: React.ChangeEvent<
      HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement
    >,
  ) => void;
  handleSubmit: (e: React.SubmitEvent<HTMLFormElement>) => Promise<void>;
  reset: (nextValues?: T) => void;
  validateField: (field: keyof T) => void;
}

export function useZodForm<T>({
  initialValues,
  schema,
  onSubmit,
}: UseZodFormOptions<T>): UseZodFormResult<T> {
  const [values, setValues] = useState<T>(initialValues);
  const [errors, setErrors] = useState<FieldErrors<T>>({});
  const [touched, setTouchedState] = useState<
    Partial<Record<keyof T, boolean>>
  >({});
  const [submitting, setSubmitting] = useState(false);
  const [serverError, setServerError] = useState("");

  const setValue = useCallback(<K extends keyof T>(field: K, value: T[K]) => {
    setValues((prev) => ({ ...prev, [field]: value }));
    setErrors((prev) => ({ ...prev, [field]: undefined }));
  }, []);

  const handleChange = useCallback(
    (
      e: React.ChangeEvent<
        HTMLInputElement | HTMLSelectElement | HTMLTextAreaElement
      >,
    ) => {
      const { name, value, type } = e.target;
      const v =
        type === "number" ? (value === "" ? undefined : Number(value)) : value;
      setValue(name as keyof T, v as T[keyof T]);
    },
    [setValue],
  );

  const setTouched = useCallback((field: keyof T) => {
    setTouchedState((prev) => ({ ...prev, [field]: true }));
  }, []);

  const validateField = useCallback(
    (field: keyof T) => {
      const parsed = schema.safeParse(values);
      if (!parsed.success) {
        const allErrors = zodToFieldErrors(parsed.error);
        setErrors((prev) => ({ ...prev, [field]: allErrors[field] }));
      } else {
        setErrors((prev) => ({ ...prev, [field]: undefined }));
      }
    },
    [schema, values],
  );

  const handleSubmit = useCallback(
    async (e: React.SubmitEvent<HTMLFormElement>) => {
      e.preventDefault();
      setServerError("");
      const parsed = schema.safeParse(values);
      if (!parsed.success) {
        setErrors(zodToFieldErrors(parsed.error));
        // Mark all as touched so errors show
        const allTouched: Partial<Record<keyof T, boolean>> = {};
        for (const k of Object.keys(values as object)) {
          allTouched[k as keyof T] = true;
        }
        setTouchedState(allTouched);
        return;
      }
      setSubmitting(true);
      try {
        await onSubmit(parsed.data);
      } catch (err) {
        setServerError(
          err instanceof Error ? err.message : "Submission failed",
        );
      } finally {
        setSubmitting(false);
      }
    },
    [schema, values, onSubmit],
  );

  const reset = useCallback(
    (nextValues?: T) => {
      setValues(nextValues ?? initialValues);
      setErrors({});
      setTouchedState({});
      setServerError("");
    },
    [initialValues],
  );

  return useMemo(
    () => ({
      values,
      errors,
      touched,
      submitting,
      serverError,
      setValue,
      setTouched,
      handleChange,
      handleSubmit,
      reset,
      validateField,
    }),
    [
      values,
      errors,
      touched,
      submitting,
      serverError,
      setValue,
      setTouched,
      handleChange,
      handleSubmit,
      reset,
      validateField,
    ],
  );
}

export function useMyAppointments() {
  const { data, error, isLoading, mutate } = useSWR(MINE_KEY, () =>
    fetcher(() => appointmentsApi.mine()),
  );

  const invalidate = () => globalMutate(MINE_KEY);

  const create = async (payload: AppointmentCreateDTO) => {
    const created = await appointmentsApi.create(payload);
    await invalidate();
    return created;
  };

  const cancel = async (id: string) => {
    // Optimistic
    await mutate(
      (current) =>
        current?.map((a) =>
          String(a.id) === id ? { ...a, status: "CANCELLED" as const } : a,
        ),
      { revalidate: false },
    );
    try {
      const updated = await appointmentsApi.cancel(id);
      await invalidate();
      return updated;
    } catch (e) {
      await invalidate();
      throw e;
    }
  };

  const complete = async (id: string) => {
    await mutate(
      (current) =>
        current?.map((a) =>
          String(a.id) === id ? { ...a, status: "COMPLETED" as const } : a,
        ),
      { revalidate: false },
    );
    try {
      const updated = await appointmentsApi.complete(id);
      await invalidate();
      return updated;
    } catch (e) {
      await invalidate();
      throw e;
    }
  };

  const addNotes = async (id: string, notes: string) => {
    const updated = await appointmentsApi.addNotes(id, {notes});
    await invalidate();
    return updated;
  };

  return {
    appointments: data ?? [],
    error,
    isLoading,
    create,
    cancel,
    complete,
    addNotes,
    invalidate,
  };
}

const key = (filters: UserFilters = {}) => ["admin/users", filters] as const;

export function useUsers(filters: UserFilters = {}) {
  const { data, error, isLoading } = useSWR(key(filters), () =>
    fetcher(() => usersApi.list(filters)),
  );

  const invalidate = () =>
    mutate((k) => Array.isArray(k) && k[0] === "admin/users");

  const create = async (payload: UserCreateDTO) => {
    const created = await usersApi.create(payload);
    await invalidate();
    return created;
  };

  const update = async (id: string, payload: UserUpdateDTO) => {
    const updated = await usersApi.update(id, payload);
    await invalidate();
    return updated;
  };

  const setEnabled = async (id: string, enabled: boolean) => {
    // optimistic
    await mutate(
      (k) => Array.isArray(k) && k[0] === "admin/users",
      (current: UserResponseDTO[] | undefined) =>
        current?.map((u) => (u.id === id ? { ...u, enabled } : u)),
      { revalidate: false },
    );
    try {
      const updated = await usersApi.setEnabled(id, enabled);
      await invalidate();
      return updated;
    } catch (e) {
      await invalidate();
      throw e;
    }
  };

  const remove = async (id: string) => {
    // Optimistic remove
    await mutate(
      (k) => Array.isArray(k) && k[0] === "admin/users",
      (current: UserResponseDTO[] | undefined) => current?.filter((u) => u.id !== id),
      { revalidate: false },
    );
    try {
      await usersApi.remove(id);
    } finally {
      await invalidate();
    }
  };

  return {
    users: data ?? [],
    error,
    isLoading,
    create,
    update,
    setEnabled,
    remove,
    invalidate,
  };
}

import { createContext, useContext, useState, useCallback, type ReactNode } from 'react';
import { authApi, HttpError } from '../lib/api';
import type { CurrentUser, LoginRequestDTO, PatientCreateDTO } from '../types';

interface AuthContextValue {
  user: CurrentUser | null;
  login: (payload: LoginRequestDTO) => Promise<void>;
  register: (payload: PatientCreateDTO) => Promise<void>;
  logout: () => Promise<void>;
  isAuthenticated: boolean;
  refresh: () => Promise<void>;
}

const AuthContext = createContext<AuthContextValue | null>(null);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<CurrentUser | null>(() => {
    const raw = sessionStorage.getItem('medicore_user');
    return raw ? (JSON.parse(raw) as CurrentUser) : null;
  });

  const persist = (u: CurrentUser | null) => {
    setUser(u);
    if (u) sessionStorage.setItem('medicore_user', JSON.stringify(u));
    else sessionStorage.removeItem('medicore_user');
  };

  const login = useCallback(async (payload: LoginRequestDTO) => {
    // authApi.login triggers Basic auth via GET /api/auth/me
    const current = await authApi.login(payload);
    persist(current);
  }, []);

  const register = useCallback(async (payload: PatientCreateDTO) => {
    // 1) register (public) 2) then authenticate with the same credentials
    await authApi.register(payload);
    const current = await authApi.login({ email: payload.email, password: payload.password });
    persist(current);
  }, []);

  const logout = useCallback(async () => {
    await authApi.logout();
    persist(null);
  }, []);

  const refresh = useCallback(async () => {
    try {
      const current = await authApi.me();
      persist(current);
    } catch (e) {
      if (e instanceof HttpError && e.status === 401) persist(null);
      else throw e;
    }
  }, []);

  return (
    <AuthContext.Provider
      value={{ user, login, register, logout, isAuthenticated: !!user, refresh }}
    >
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth must be used within AuthProvider');
  return ctx;
