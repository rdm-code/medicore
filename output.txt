import com.medicore.hospitalmangement.entities.AppointmentEntity;
import com.medicore.hospitalmangement.entities.AppointmentStatus;
import org.springframework.data.jpa.repository.*;
import org.springframework.data.repository.query.Param;

import java.time.LocalDateTime;
import java.util.List;
import java.util.Optional;

public interface AppointmentsRepo extends JpaRepository<AppointmentEntity, String> {

    // join fetch avoids N+1 queries when the mapper reads doctor and patient names
    @Query("""
           select a from AppointmentEntity a
           join fetch a.doctor join fetch a.patient
           where a.id = :id
           """)
    Optional<AppointmentEntity> findByIdWithDetails(@Param("id") String id);

    @Query("""
           select a from AppointmentEntity a
           join fetch a.doctor join fetch a.patient p
           where p.user.email = :email
           order by a.appointmentTime
           """)
    List<AppointmentEntity> findByPatientEmail(@Param("email") String email);

    @Query("""
           select a from AppointmentEntity a
           join fetch a.doctor d join fetch a.patient
           where d.user.email = :email
           order by a.appointmentTime
           """)
    List<AppointmentEntity> findByDoctorEmail(@Param("email") String email);

    @Query("""
           select a from AppointmentEntity a
           join fetch a.doctor d join fetch a.patient
           where (:status is null or a.status = :status)
             and (:doctorId is null or d.id = :doctorId)
           order by a.appointmentTime
           """)
    List<AppointmentEntity> search(@Param("status") AppointmentStatus status,
                                   @Param("doctorId") String doctorId);

    boolean existsByDoctorIdAndAppointmentTimeAndStatusNot(
            String doctorId, LocalDateTime time, AppointmentStatus status);

    /**
     * All SCHEDULED appointment times for a doctor within [dayStart, dayEnd).
     * Used to compute available slots.
     */
    @Query("""
        select a.appointmentTime
          from AppointmentEntity a
         where a.doctor.id = :doctorId
           and a.appointmentTime >= :dayStart
           and a.appointmentTime <  :dayEnd
           and a.status = :status
        """)
    List<LocalDateTime> findScheduledTimesForDoctorOnDay(
            @Param("doctorId") String doctorId,
            @Param("dayStart") LocalDateTime dayStart,
            @Param("dayEnd")   LocalDateTime dayEnd,
            @Param("status")   AppointmentStatus status
    );

            @Valid @RequestBody LoginRequestDTO dto,
            HttpServletRequest request, HttpServletResponse response) {

        System.out.println("fjasfjdoiafd");
        // Calls CustomUserDetailsService + BCrypt check; throws on failure
        Authentication auth = authManager.authenticate(
                new UsernamePasswordAuthenticationToken(dto.email(), dto.password()));

        SecurityContext context = SecurityContextHolder.createEmptyContext();
        context.setAuthentication(auth);
        SecurityContextHolder.setContext(context);
        securityContextRepository.saveContext(context, request, response);  // creates JSESSIONID

        String role = Objects.requireNonNull(auth.getAuthorities().iterator().next()
                .getAuthority()).replace("ROLE_", "");
        return ResponseEntity.ok(Map.of("email", auth.getName(), "role", role));
    }

    public ResponseEntity<Map<String, String>> me(Authentication auth) {

@Configuration
@EnableMethodSecurity
public class SecurityConfig {

    private final ObjectMapper mapper;

    public SecurityConfig(ObjectMapper mapper) {
        this.mapper = mapper;
    }

    @Bean
    SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
                .cors(Customizer.withDefaults())
                .csrf(AbstractHttpConfigurer::disable)
                .securityContext(sc -> sc.requireExplicitSave(true))
                .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.IF_REQUIRED))
                .authorizeHttpRequests(auth -> auth
                        .requestMatchers("/api/auth/login", "/api/auth/register",
                                "/swagger-ui/**", "/v3/api-docs/**",
                                "/actuator/health", "/actuator/health/**").permitAll()   // added for Docker healthcheck
                        .requestMatchers("/api/admin/**").hasRole("ADMIN")
                        .anyRequest().authenticated())

                // No formLogin(): login is handled by AuthController (JSON body)
                .logout(logout -> logout
                        .logoutUrl("/api/auth/logout")
                        .logoutSuccessHandler((req, res, auth) ->
                                res.setStatus(HttpStatus.NO_CONTENT.value()))
                        .deleteCookies("JSESSIONID"))

                .httpBasic(Customizer.withDefaults())

                .exceptionHandling(e -> e
                        .authenticationEntryPoint((req, res, ex) ->
                                writeError(res, HttpStatus.UNAUTHORIZED,
                                        "Authentication required", req.getRequestURI()))
                        .accessDeniedHandler((req, res, ex) ->
                                writeError(res, HttpStatus.FORBIDDEN,
                                        "You do not have permission to do this", req.getRequestURI())));

        return http.build();
    }

    @Bean
    PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }

    @Bean
    AuthenticationManager authenticationManager(AuthenticationConfiguration cfg) throws Exception {
        return cfg.getAuthenticationManager();
    }

    @Bean
    SecurityContextRepository securityContextRepository() {
        return new HttpSessionSecurityContextRepository();
    }

    @Bean
    CorsConfigurationSource corsConfigurationSource() {
        var c = new CorsConfiguration();

        // Allow both dev (Vite :5173) and Docker (Nginx :80) origins.
        // Add your production hostname here when you deploy.
        c.setAllowedOrigins(List.of(
                "http://localhost:5173",
                "http://localhost:3000",
                "http://localhost",          // Nginx frontend in Docker on port 80
                "http://127.0.0.1"
        ));

        c.setAllowedMethods(List.of("GET","POST","PUT","PATCH","DELETE","OPTIONS"));
        c.setAllowCredentials(true);
        c.setAllowedHeaders(List.of("*"));
        c.setExposedHeaders(List.of("Set-Cookie"));

        var s = new UrlBasedCorsConfigurationSource();
        s.registerCorsConfiguration("/**", c);
        return s;
    }

    private void writeError(HttpServletResponse res, HttpStatus status,
                            String message, String path) throws IOException {
        res.setStatus(status.value());
        res.setContentType(MediaType.APPLICATION_JSON_VALUE);
        mapper.writeValue(res.getOutputStream(), Map.of(
                "timestamp", Instant.now().toString(),
                "status", status.value(),
                "error", status.getReasonPhrase(),
                "message", message,
                "path", path));
    }

@RestControllerAdvice
@Order(Ordered.LOWEST_PRECEDENCE)
public class GlobalExceptionHandler {

    // Safety net: never leak stack traces to the client
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ErrorResponse> handleOther(Exception ex, HttpServletRequest req) {
        System.out.println("Global : "+ex.getClass().getName());
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "Something went wrong", req, null);
    }

    private ResponseEntity<ErrorResponse> build(HttpStatus status, String message,
                                                HttpServletRequest req,
                                                Map<String, String> fieldErrors) {
        ErrorResponse body = new ErrorResponse(
                Instant.now(), status.value(), status.getReasonPhrase(),
                message, req.getRequestURI(), fieldErrors);
        return ResponseEntity.status(status).body(body);
    }

@RestControllerAdvice
@Order(Ordered.HIGHEST_PRECEDENCE)
public class ValidationExceptionHandler {

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String,String>> handleMethodArgumentNotValidException(MethodArgumentNotValidException ex){
        Map<String,String> map = new HashMap<>();
        ex.getBindingResult().getAllErrors().forEach((error)->{
            map.put(((FieldError) error).getField(),error.getDefaultMessage());
        });

        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(map);
    }
    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ErrorResponse> handleDataIntegrityViolation(DataIntegrityViolationException ex, HttpServletRequest req) {
        String message = "A database constraint violation occurred.";

        // Check if the violation is related to the email column/index
        if (ex.getMessage() != null && ex.getMessage().toUpperCase().contains("EMAIL")) {
            message = "Email address already exists.";
        }

        return build(HttpStatus.CONFLICT,message,req,null);
    }

    private ResponseEntity<ErrorResponse> build(HttpStatus status, String message,
                                                HttpServletRequest req,
                                                Map<String, String> fieldErrors) {
        ErrorResponse body = new ErrorResponse(
                Instant.now(), status.value(), status.getReasonPhrase(),
                message, req.getRequestURI(), fieldErrors);
        return ResponseEntity.status(status).body(body);
    }
}

@RestControllerAdvice

        Map<String, Object> body = Map.of(
                "timestamp", Instant.now(),
                "status", HttpStatus.CONFLICT.value(),
                "error", "Conflict",
                "message", ex.getMessage()
        );

        return ResponseEntity.status(HttpStatus.CONFLICT).body(body);
    }

    @ExceptionHandler(ResourceNotFoundException.class)

        Map<String, Object> body = Map.of(
                "timestamp", Instant.now(),
                "status", HttpStatus.NOT_FOUND.value(),
                "error", "Not Found",
                "message", ex.getMessage()
        );

        return ResponseEntity.status(HttpStatus.NOT_FOUND).body(body);
    }

        return build(HttpStatus.CONFLICT, ex.getMessage(), req, null);

    public ResponseEntity<ErrorResponse> handleBadTransition(InvalidStatusTransitionException ex, HttpServletRequest req) {
        return build(HttpStatus.CONFLICT, ex.getMessage(), req, null);

@Entity
@Table(name = "users")
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class UserEntity {
    @Id @GeneratedValue(strategy = GenerationType.UUID)
    String id;

    @Column(nullable = false, unique = true)
    String email;

    @Column(nullable = false)
    String password;

    @Column(nullable = true)
    String displayName;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private Role role;

    @Column(name = "is_enabled", nullable = false)
//    @ColumnDefault("true")
    @Builder.Default
    private boolean enabled = true;

    @CreationTimestamp
    @Column(updatable = false)
    private Instant createdAt;

    @UpdateTimestamp
    private Instant updatedAt;

    @Override
    public String toString() {
        return "UserEntity{" +
                "id='" + id + '\'' +
                ", email='" + email + '\'' +
                ", password='" + password + '\'' +
                ", displayName='" + displayName + '\'' +
                ", role=" + role +
                ", enabled=" + enabled +
                ", createdAt=" + createdAt +
                ", updatedAt=" + updatedAt +
                '}';
    }
}

@Service
@Transactional(readOnly = true)
public class AppointmentsService {

    private static final int SLOT_MINUTES = 30;

    private final AppointmentsRepo repo;
    private final PatientsRepo patientsRepo;
    private final DoctorsRepo doctorsRepo;
    private final AppointmentMapper mapper;

    public AppointmentsService(AppointmentsRepo repo, PatientsRepo patientsRepo,
                               DoctorsRepo doctorsRepo, AppointmentMapper mapper) {
        this.repo = repo;
        this.patientsRepo = patientsRepo;
        this.doctorsRepo = doctorsRepo;
        this.mapper = mapper;
    }

    // ---------- Book (PATIENT) ----------

    @Transactional
    public AppointmentResponseDTO book(String patientEmail, AppointmentCreateDTO dto) {
        PatientEntity patient = patientsRepo.findByUserEmail(patientEmail)
                .orElseThrow(() -> new ResourceNotFoundException("Patient profile for " + patientEmail));

        // Row lock on the doctor: two simultaneous bookings for the same doctor run one after the other
        DoctorEntity doctor = doctorsRepo.findByIdForUpdate(dto.doctorId())
                .orElseThrow(() -> new ResourceNotFoundException("Doctor", dto.doctorId()));

        validateSlot(dto.appointmentTime());

        if (repo.existsByDoctorIdAndAppointmentTimeAndStatusNot(
                doctor.getId(), dto.appointmentTime(), AppointmentStatus.CANCELLED)) {
            throw new SlotAlreadyBookedException("Doctor is already booked at " + dto.appointmentTime());
        }

        AppointmentEntity appointment = mapper.toEntity(dto);
        appointment.setPatient(patient);
        appointment.setDoctor(doctor);
        return mapper.toResponseDTO(repo.save(appointment));
    }

    // ---------- Read ----------

    public List<AppointmentResponseDTO> getAll(AppointmentStatus status, String doctorId) {
        return repo.search(status, doctorId).stream().map(mapper::toResponseDTO).toList();
    }

    public List<AppointmentResponseDTO> getMine(String email, boolean isDoctor) {
        List<AppointmentEntity> list = isDoctor
                ? repo.findByDoctorEmail(email)
                : repo.findByPatientEmail(email);
        return list.stream().map(mapper::toResponseDTO).toList();
    }

    public AppointmentResponseDTO getById(String id, String email, boolean isAdmin) {
        AppointmentEntity a = findOrThrow(id);
        requireParticipantOrAdmin(a, email, isAdmin);
        return mapper.toResponseDTO(a);
    }

    // ---------- Status changes ----------

    @Transactional
    public AppointmentResponseDTO cancel(String id, String email, boolean isAdmin) {
        AppointmentEntity a = findOrThrow(id);
        requireParticipantOrAdmin(a, email, isAdmin);
        transition(a, AppointmentStatus.CANCELLED);
        return mapper.toResponseDTO(a);
    }

    @Transactional
    public AppointmentResponseDTO complete(String id, String email, boolean isAdmin) {
        AppointmentEntity a = findOrThrow(id);
        if (!isAdmin && !isDoctorOf(a, email)) {
            throw new AccessDeniedException("Only the assigned doctor can complete this appointment");
        }
        transition(a, AppointmentStatus.COMPLETED);
        return mapper.toResponseDTO(a);
    }

    @Transactional
    public AppointmentResponseDTO addNotes(String id, String doctorEmail, AppointmentNotesDTO dto) {
        AppointmentEntity a = findOrThrow(id);
        if (!isDoctorOf(a, doctorEmail)) {
            throw new AccessDeniedException("Only the assigned doctor can add notes");
        }
        if (a.getStatus() == AppointmentStatus.CANCELLED) {
            throw new InvalidStatusTransitionException("Cannot add notes to a cancelled appointment");
        }
        a.setNotes(dto.notes().trim());
        return mapper.toResponseDTO(a);
    }

    // ---------- Delete (ADMIN) ----------

    @Transactional
    public void delete(String id) {
        repo.delete(findOrThrow(id));
    }

    // ---------- Helpers ----------

    private AppointmentEntity findOrThrow(String id) {
        return repo.findByIdWithDetails(id)
                .orElseThrow(() -> new ResourceNotFoundException("Appointment", id));
    }

    /** Only SCHEDULED appointments may change; COMPLETED and CANCELLED are final. */
    private void transition(AppointmentEntity a, AppointmentStatus target) {
        if (a.getStatus() != AppointmentStatus.SCHEDULED) {
            throw new InvalidStatusTransitionException(
                    "Cannot change a " + a.getStatus() + " appointment to " + target);
        }
        a.setStatus(target);
    }

    /** Slots start on the hour or half hour, e.g. 10:00 or 10:30. */
    private void validateSlot(LocalDateTime time) {
        if (time.getMinute() % SLOT_MINUTES != 0 || time.getSecond() != 0 || time.getNano() != 0) {
            throw new IllegalArgumentException(
                    "Appointments must start on a " + SLOT_MINUTES + "-minute boundary (e.g. 10:00, 10:30)");
        }
    }

    private boolean isPatientOf(AppointmentEntity a, String email) {
        return a.getPatient().getUser().getEmail().equals(email);
    }

    private boolean isDoctorOf(AppointmentEntity a, String email) {
        return a.getDoctor().getUser().getEmail().equals(email);
    }

    private void requireParticipantOrAdmin(AppointmentEntity a, String email, boolean isAdmin) {
        if (!isAdmin && !isPatientOf(a, email) && !isDoctorOf(a, email)) {
            throw new AccessDeniedException("You do not have access to this appointment");
        }
    }

@Service

    private final UsersRepo usersRepo;

    public CustomUserDetailsService(UsersRepo usersRepo) {
        this.usersRepo = usersRepo;
    }

    @Override
    public UserDetails loadUserByUsername(String email) {
        UserEntity u = usersRepo.findByEmail(email)          // <-- database query

                .password(u.getPassword())                   // BCrypt hash from the DB
                .authorities("ROLE_" + u.getRole().name())
                .disabled(!u.isEnabled())
                .build();
    }

@Service
public class SlotsService {

    private static final LocalTime OPEN_TIME  = LocalTime.of(8, 0);
    private static final LocalTime CLOSE_TIME = LocalTime.of(16, 0);
    private static final int SLOT_MINUTES = 30;

    private static final DateTimeFormatter HH_MM = DateTimeFormatter.ofPattern("HH:mm");

    private final AppointmentsRepo appointmentsRepo;
    private final DoctorsRepo doctorRepo;

    public SlotsService(AppointmentsRepo appointmentsRepo, DoctorsRepo doctorRepo) {
        this.appointmentsRepo = appointmentsRepo;
        this.doctorRepo = doctorRepo;
    }

    @Transactional(readOnly = true)
    public List<String> getAvailableSlots(String doctorId, LocalDate date) {
        if (!doctorRepo.existsById(doctorId)) {
            throw new ResourceNotFoundException("Doctor",doctorId);
        }

        // Only today or future dates
        if (date.isBefore(LocalDate.now())) {
            return List.of();
        }

        LocalDateTime dayStart = date.atStartOfDay();
        LocalDateTime dayEnd   = date.plusDays(1).atStartOfDay();

        List<LocalDateTime> booked = appointmentsRepo.findScheduledTimesForDoctorOnDay(
                doctorId, dayStart, dayEnd, AppointmentStatus.SCHEDULED
        );

        Set<LocalTime> bookedTimes = new HashSet<>();
        for (LocalDateTime dt : booked) {
            bookedTimes.add(dt.toLocalTime());
        }

        List<String> available = new ArrayList<>();
        LocalTime cursor = OPEN_TIME;
        while (cursor.isBefore(CLOSE_TIME)) {
            if (!bookedTimes.contains(cursor)) {
                available.add(cursor.format(HH_MM));
            }
            cursor = cursor.plusMinutes(SLOT_MINUTES);
        }
        return available;
    }

@Service
@Transactional(readOnly = true)   // default for reads; write methods override below
public class PatientsService {

    private final PatientsRepo repo;
    private final UsersRepo usersRepo;
    private final PatientMapper mapper;
    private final PasswordEncoder passwordEncoder;

    public PatientsService(PatientsRepo repo, UsersRepo usersRepo,
                           PatientMapper mapper, PasswordEncoder passwordEncoder) {
        this.repo = repo;
        this.usersRepo = usersRepo;
        this.mapper = mapper;
        this.passwordEncoder = passwordEncoder;
    }

    // ---------- Read ----------

    public List<PatientResponseDTO> getAllPatients() {
        return repo.findAll().stream().map(mapper::toResponseDTO).toList();
    }

    public PatientResponseDTO getPatientById(String id) {
        return mapper.toResponseDTO(findOrThrow(id));
    }

    /** For GET /api/patients/me: the patient is derived from the logged-in email. */
    public PatientResponseDTO getMyProfile(String email) {
        PatientEntity patient = repo.findByUserEmail(email)
                .orElseThrow(() -> new ResourceNotFoundException("Patient profile for " + email));
        return mapper.toResponseDTO(patient);
    }

    // ---------- Create (self-registration) ----------

    @Transactional
    public PatientResponseDTO createPatient(PatientCreateDTO dto) {
        if (usersRepo.existsByEmail(dto.email())) {
            throw new DuplicateResourceException("Email address already exists.");
        }

        UserEntity user = UserEntity.builder()
                .email(dto.email())
                .displayName(dto.fullName())                       // remove if your entity has no such field
                .password(passwordEncoder.encode(dto.password()))  // always hash
                .role(Role.PATIENT)                                // fixed server-side, never from the client
                .enabled(true)
                .build();
        usersRepo.save(user);

        PatientEntity patient = mapper.toEntity(dto);
        patient.setUser(user);
        return mapper.toResponseDTO(repo.save(patient));
    }

    // ---------- Update ----------

    @Transactional
    public PatientResponseDTO updatePatient(String id, PatientUpdateDTO dto) {
        PatientEntity patient = findOrThrow(id);
        mapper.updateEntity(dto, patient);           // in-place; dirty checking issues the UPDATE
        return mapper.toResponseDTO(patient);
    }

    @Transactional
    public PatientResponseDTO updateMyProfile(String email, PatientUpdateDTO dto) {
        PatientEntity patient = repo.findByUserEmail(email)
                .orElseThrow(() -> new ResourceNotFoundException("Patient profile for " + email));
        mapper.updateEntity(dto, patient);
        return mapper.toResponseDTO(patient);
    }

    // ---------- Delete ----------

    @Transactional
    public void deletePatientById(String id) {
        PatientEntity patient = findOrThrow(id);
        UserEntity user = patient.getUser();
        repo.delete(patient);        // profile first (it holds the FK)
        usersRepo.delete(user);      // then the login account
    }

    // ---------- Helper ----------

    private PatientEntity findOrThrow(String id) {
        return repo.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Patient", id));
    }

@Service
public class UsersService {

    UsersRepo repo;
    UserMapper mapper;
    private final PasswordEncoder passwordEncoder;

    public UsersService(UsersRepo repo, UserMapper mapper, PasswordEncoder passwordEncoder) {
        this.repo = repo;
        this.mapper = mapper;
        this.passwordEncoder = passwordEncoder;
    }

    public List<UserResponseDTO> getUsers() {
        //        System.out.println(users);
        return repo.findAll().stream().map(mapper::toResponseDTO).toList();
    }

    public UserResponseDTO getUserById(String id) {
        return mapper.toResponseDTO(findOrThrow(id));
    }

    @Transactional
    public UserResponseDTO createUser(UserCreateDTO dto) {
        if (repo.existsByEmail(dto.email())) {
            throw new DuplicateResourceException("Email address already exists.");
        }
        UserEntity user = mapper.toEntity(dto);
        user.setPassword(passwordEncoder.encode(dto.password()));   // always hash
        return mapper.toResponseDTO(repo.save(user));
    }

    @Transactional
    public UserResponseDTO updateUser(String id, UserUpdateDTO dto) {
        UserEntity user = repo.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("User", id));

        if (dto.email() != null && !dto.email().equals(user.getEmail())) {
            if (repo.existsByEmail(dto.email())) {
                throw new DuplicateResourceException("Email address already exists.");
            }
            user.setEmail(dto.email());
        }

        if (dto.password() != null && !dto.password().isBlank()) {
            user.setPassword(passwordEncoder.encode(dto.password()));
        }

        mapper.updateEntity(dto, user);   // displayName, enabled, etc.
        return mapper.toResponseDTO(user);
    }

    @Transactional
    public void deleteUser(String id) {
        repo.delete(findOrThrow(id));
    }

    public boolean existsByEmail(String email) {
        return repo.existsByEmail(email);
    }

    private UserEntity findOrThrow(String id) {
        return repo.findById(id).orElseThrow(() -> new ResourceNotFoundException("User", id));
    }}
